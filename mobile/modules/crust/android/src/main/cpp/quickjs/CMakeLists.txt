cmake_minimum_required(VERSION 3.22.1)
project(mentra_quickjs C)
set(CMAKE_C_STANDARD 99)
file(READ "engine/VERSION" QUICKJS_VERSION)
string(STRIP "${QUICKJS_VERSION}" QUICKJS_VERSION)
file(GLOB JNI_SOURCES CONFIGURE_DEPENDS "jni/*.c" "jni/mapping/*.c")
add_library(quickjs SHARED
  engine/cutils.c engine/libbf.c engine/libregexp.c engine/libunicode.c
  engine/quickjs.c common/quickjs_version.c ${JNI_SOURCES})
target_include_directories(quickjs PRIVATE engine common c-vector jni jni/mapping)
target_compile_definitions(quickjs PRIVATE CONFIG_BIGNUM CONFIG_VERSION="${QUICKJS_VERSION}")
if(ANDROID)
  target_compile_definitions(quickjs PRIVATE CONFIG_ANDROID)
  target_link_libraries(quickjs PRIVATE log m)
  target_link_options(quickjs PRIVATE "-Wl,-z,max-page-size=16384")
else()
  find_package(JNI REQUIRED)
  target_include_directories(quickjs PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(quickjs PRIVATE pthread m)
endif()
